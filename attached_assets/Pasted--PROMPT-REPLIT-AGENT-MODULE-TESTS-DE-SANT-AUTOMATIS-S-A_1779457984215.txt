---

## 🎯 OBJECTIF

Ajouter à l'application Goûtstoso un **module de tests automatisés** qui permet à Jordan de vérifier en 1 clic que :

1. ✅ La compta est cohérente
2. ✅ Le stock est cohérent
3. ✅ Les automatismes fonctionnent
4. ✅ Les données sont intègres
5. ✅ L'application n'a pas de bug

Le tout sans toucher aux données réelles.

---

## ⛔ RÈGLES ABSOLUES

1. **NE PAS modifier** les données existantes lors des tests
2. **NE PAS supprimer** les composants existants
3. **TOUJOURS travailler sur des données fictives** clairement marquées (préfixe "TEST_")
4. **NETTOYER automatiquement** les données de test après chaque exécution
5. **MAINTENIR** tous les automatismes compta existants

#### Test 1.1 : Cohérence solde PostFinance
```javascript
function test_coherence_solde() {
  // Somme de toutes les écritures compta
  const totalRecettes = st.ecrituresComptables
    .filter(e => e.type === "recette")
    .reduce((sum, e) => sum + e.montant, 0);
    
  const totalDepenses = st.ecrituresComptables
    .filter(e => e.type === "depense")
    .reduce((sum, e) => sum + Math.abs(e.montant), 0);
    
  const soldeCalcule = SOLDE_INITIAL + totalRecettes - totalDepenses;
  const soldeAffiche = st.soldePostFinance;
  
  if (Math.abs(soldeCalcule - soldeAffiche) > 0.01) {
    return {
      success: false,
      message: `Solde affiché (${soldeAffiche}) ≠ solde calculé (${soldeCalcule})`,
      diagnostic: "Le solde PostFinance ne correspond pas au total des écritures.",
      solution: "Vérifier les écritures comptables manquantes ou en double."
    };
  }
  return { success: true };
}
```

function test_factures_orphelines() {
  // Une facture orpheline = facture sans client/partenaire valide
  const orphelines = st.factures.filter(f => {
    if (f.clientId) {
      return !st.clients.find(c => c.id === f.clientId);

function test_doublons_ids() {
  const erreurs = [];
  const collections = ['clients', 'partenaires', 'factures', 'produits', 'commandes'];
  
  collections.forEach(coll => {
    if (!st[coll]) return;
    const ids = st[coll].map(item => item.id);
    const doublons = ids.filter((id, idx) => ids.indexOf(id) !== idx);
    if (doublons.length > 0) {
      erreurs.push(`${coll}: doublons ${doublons.join(', ')}`);
    }
  });
  
  if (erreurs.length > 0) {
    return {
      success: false,
      message: `Doublons d'IDs détectés`,
      details: erreurs,
      solution: "Régénérer les IDs en doublon pour assurer l'unicité."
    };
  }
  return { success: true };
}

    { nom: "Performance", fn: test_performance, categorie: "Interface" }
  ];
  
  for (const test of tousLesTests) {
    try {
      const resultat = test.async ? await test.fn() : test.fn();
      resultats.tests.push({
        nom: test.nom,
        categorie: test.categorie,
        ...resultat
      });
      
      resultats.totalTests++;
      if (resultat.skipped) {
        resultats.ignores++;
      } else if (resultat.success) {
        resultats.reussis++;
      } else {
        resultats.echecs++;
      }
    } catch (e) {
      resultats.tests.push({
        nom: test.nom,
        categorie: test.categorie,
        success: false,
        message: "Exception non gérée",
        diagnostic: e.message,
        solution: "Bug critique à investiguer."
      });
      resultats.totalTests++;
      resultats.echecs++;
    }
  }
  
  // Sauvegarder les résultats
  st.derniersResultatsTests = resultats;
  setSt({ ...st });
  
  return resultats;
}

function exporterRapportPDF(resultats) {
  const pdf = new jsPDF();
  
  pdf.setFontSize(20);
  pdf.text("Rapport de tests de santé Goûtstoso", 20, 20);
  
  pdf.setFontSize(12);
  pdf.text(`Date : ${formatDate(resultats.dateExecution)}`, 20, 35);
  pdf.text(`Tests réussis : ${resultats.reussis}/${resultats.totalTests}`, 20, 45);
  pdf.text(`Erreurs : ${resultats.echecs}`, 20, 55);
  
  let y = 75;
  
  // Erreurs en premier
  if (resultats.echecs > 0) {
    pdf.setFontSize(14);
    pdf.text("ERREURS DÉTECTÉES", 20, y);
    y += 10;
    
    resultats.tests.filter(t => !t.success && !t.skipped).forEach(t => {
      pdf.setFontSize(10);
      pdf.text(`❌ ${t.nom}`, 20, y);
      y += 7;
      pdf.text(`Problème : ${t.message}`, 30, y);
      y += 5;
      if (t.solution) {
        pdf.text(`Solution : ${t.solution}`, 30, y);
        y += 8;
      }
      
      if (y > 270) {
        pdf.addPage();
        y = 20;
      }
    });
  }
  
  pdf.save(`rapport-tests-${Date.now()}.pdf`);
}

function planifierTestsHebdomadaires() {
  const dernierTest = localStorage.getItem('dernier_test_hebdo');
